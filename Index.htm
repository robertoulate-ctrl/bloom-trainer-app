    <script>

        // --- DOM Elements & Event Listeners ---

        const generateBtn = document.getElementById('generateBtn');

        const recipeOutput = document.getElementById('recipeOutput');

        const methodSelect = document.getElementById('method');

        const filterPaperSelect = document.getElementById('filterPaper');

 

        generateBtn.addEventListener('click', generateRecipe);

       

        // Disable filter paper selection for Aeropress, as it's not relevant

        methodSelect.addEventListener('change', () => {

            filterPaperSelect.disabled = methodSelect.value === 'Aeropress';

        });

 

        // --- Core Recipe Generation Logic ---

        function generateRecipe() {

            const method = methodSelect.value;

            const filterPaper = filterPaperSelect.value;

            const process = document.getElementById('process').value;

            const grams = parseFloat(document.getElementById('grams').value);

            const ratio = parseFloat(document.getElementById('ratio').value);

 

            // Handle Aeropress as a special case

            if (method === 'Aeropress') {

                generateAeropressRecipe(process, grams);

                recipeOutput.classList.remove('hidden');

                return;

            }

 

            // Calculate all recipe parameters

            const totalWater = Math.round(grams * ratio);

            const grind = getGrindSetting(method, process, grams, ratio, filterPaper);

            const temp = getWaterTemperature(process, ratio, grams);

            const pours = getPourStructure(totalWater, grams, process, ratio);

            const notes = getStrategyNotes(method, process, ratio, grind.adjustmentReason);

 

            // Display results in the UI

            document.getElementById('totalWater').textContent = `${totalWater}g`;

            document.getElementById('grindSetting').textContent = `${grind.value} clics`;

            document.getElementById('waterTemp').textContent = `${temp}°C`;

           

            const pourList = document.getElementById('pourStructure');

            pourList.innerHTML = ''; // Clear previous results

            pours.forEach(pour => {

                const li = document.createElement('li');

                li.textContent = pour;

                pourList.appendChild(li);

            });

 

            document.getElementById('strategyNotes').textContent = notes;

            recipeOutput.classList.remove('hidden'); // Show the recipe card

        }

 

        function generateAeropressRecipe(process, grams) {

            document.getElementById('totalWater').textContent = `N/A (Concentrado + Bypass)`;

            document.getElementById('grindSetting').textContent = `18 clics`;

            document.getElementById('waterTemp').textContent = `90°C`;

           

            const pourList = document.getElementById('pourStructure');

            pourList.innerHTML = '';

            const steps = [

                `Método Invertido.`,

                `Añadir ${grams}g de café.`,

                `Verter ${grams * 2.5}g de agua en 15s.`,

                `Agitar suavemente por 10s.`,

                `Dejar en inmersión hasta 1:30.`,

                `Presionar suavemente por 30s.`,

                `Añadir agua (bypass) hasta alcanzar 150ml en la taza de servicio.`

            ];

            steps.forEach(step => {

                const li = document.createElement('li');

                li.textContent = step;

                pourList.appendChild(li);

            });

 

            document.getElementById('strategyNotes').textContent = `El Aeropress anula la rifa del ratio y papel, dándote máximo control. Esta receta crea un concentrado equilibrado y usa bypass para alcanzar el volumen de servicio, una táctica segura y consistente.`;

        }

 

        // Calculates a single, precise grind value based on all variables

        function getGrindSetting(method, process, grams, ratio, filterPaper) {

            let baseClicks = { V60: 24, Origami: 26, SOLO: 25, UFO: 26, April: 28, Graycano: 25 };

            let adjustment = 0;

            let reasons = [];

 

            // Adjust based on filter paper speed

            if (filterPaper === 'fast') { adjustment -= 2; reasons.push("papel de flujo rápido"); }

            else if (filterPaper === 'ultrafast') { adjustment -= 4; reasons.push("papel de flujo ultra-rápido"); }

 

            // Adjust based on coffee process

            if (process === 'Lavado') { adjustment -= 2; reasons.push("proceso lavado"); }

            if (process === 'Natural') { adjustment += 3; reasons.push("proceso natural"); }

           

            // Adjust based on dose size

            if (grams < 14) { adjustment -= 1; reasons.push("dosis baja"); }

            if (grams > 20) { adjustment += 2; reasons.push("dosis alta"); }

           

            // Adjust based on brew ratio

            if (ratio < 13) { adjustment -= 2; reasons.push("ratio corto"); }

            if (ratio > 18) { adjustment += 2; reasons.push("ratio largo"); }

           

            const finalClick = baseClicks[method] + adjustment;

            const adjustmentReason = "Punto de partida ajustado por: " + reasons.join(', ') + ".";

 

            return { value: finalClick, adjustmentReason: adjustmentReason };

        }

 

        // Calculates a single, precise temperature value

        function getWaterTemperature(process, ratio, grams) {

            let baseTemp = { Lavado: 95, Honey: 92, Natural: 90 };

            let temp = baseTemp[process];

 

            if (ratio > 18) { temp -= 2; }

            if (ratio < 13) { temp += 1; }

            if (grams > 20) { temp -= 1; }

            if (grams < 14) { temp += 1; }

           

            return temp;

        }

 

        // Formats total seconds into M:SS format

        function formatTime(totalSeconds) {

            const minutes = Math.floor(totalSeconds / 60);

            const seconds = totalSeconds % 60;

            return `${minutes}:${String(seconds).padStart(2, '0')}`;

        }

 

        // Generates the pouring structure steps

        function getPourStructure(totalWater, grams, process, ratio) {

            let pours = [];

            const bloomWater = Math.min(grams * 3, 60);

            let waterRemaining = totalWater - bloomWater;

            pours.push(`Bloom: ${bloomWater}g de agua. Esperar 45s.`);

           

            let numPours;

            if (ratio < 13) { numPours = 4; }

            else if (totalWater > 300) { numPours = 3; }

            else { numPours = 3; }

           

            if (waterRemaining <= 0) return pours;

           

            const pourSize = Math.round(waterRemaining / numPours);

 

            for (let i = 1; i <= numPours; i++) {

                const pourTimeInSeconds = 45 + (i * 30);

                const pourTarget = bloomWater + (i * pourSize);

                if (pourTarget <= totalWater) {

                   pours.push(`En ${formatTime(pourTimeInSeconds)}, verter hasta ${Math.min(pourTarget, totalWater)}g.`);

                }

            }

           

            let totalTime = "2:30 - 3:15";

            if (process === 'Natural' || ratio > 18) totalTime = "2:15 - 2:45";

            if (process === 'Lavado' && ratio < 13) totalTime = "3:00 - 3:45";

            pours.push(`Tiempo total de preparación objetivo: ${totalTime}.`);

            return pours;

        }

 

        // Generates strategic notes based on the recipe

        function getStrategyNotes(method, process, ratio, adjustmentReason) {

            let baseNote = "";

            if (process === 'Lavado') { baseNote = "Objetivo: Resaltar acidez brillante y claridad. Buscar extracción alta y uniforme para desarrollar dulzor."; }

            else if (process === 'Honey') { baseNote = "Objetivo: Equilibrio. Buscar una taza dulce, con buen cuerpo y complejidad frutal, evitando sabores fangosos."; }

            else { baseNote = "Objetivo: Controlar la intensidad. Buscar una taza jugosa y compleja, evitando sobre-extracción y sabores fermentados."; }

            if (ratio > 18) { baseNote += " Con ratio alto, el control de sobre-extracción es CRÍTICO. Considera bajar la temperatura en el último vertido."; }

            else if (ratio < 13) { baseNote += " Con ratio corto, el riesgo es la sub-extracción. Molienda fina y alta temperatura son claves."; }

            return `${adjustmentReason} ${baseNote}`;

        }

 

        // --- COMPETITION TIMER LOGIC ---

        const timerDisplay = document.getElementById('timerDisplay');

        const startStopBtn = document.getElementById('startStopBtn');

        const resetBtn = document.getElementById('resetBtn');

 

        let timerInterval = null;

        let seconds = 0;

        const MAX_SECONDS = 8 * 60; // 8 minutes competition time

        let isRunning = false;

 

        function updateTimerDisplay() {

            const minutes = Math.floor(seconds / 60);

            const remainingSeconds = seconds % 60;

            timerDisplay.textContent = `${String(minutes).padStart(2, '0')}:${String(remainingSeconds).padStart(2, '0')}`;

           

            // Change color in the last 30 seconds as a warning

            if (seconds >= MAX_SECONDS - 30) {

                timerDisplay.classList.remove('text-yellow-400');

                timerDisplay.classList.add('text-red-500');

            } else {

                timerDisplay.classList.add('text-yellow-400');

                timerDisplay.classList.remove('text-red-500');

            }

        }

 

        function startTimer() {

            if (isRunning) return;

            isRunning = true;

            startStopBtn.textContent = 'Detener';

            startStopBtn.classList.replace('btn-primary', 'btn-secondary');

 

            timerInterval = setInterval(() => {

                seconds++;

                updateTimerDisplay();

                if (seconds >= MAX_SECONDS) {

                    stopTimer();

                }

            }, 1000);

        }

 

        function stopTimer() {

            if (!isRunning) return;

            isRunning = false;

            startStopBtn.textContent = 'Iniciar';

            startStopBtn.classList.replace('btn-secondary', 'btn-primary');

            clearInterval(timerInterval);

        }

       

        function resetTimer() {

            stopTimer();

            seconds = 0;

            updateTimerDisplay();

        }

 

        startStopBtn.addEventListener('click', () => {

            if (isRunning) {

                stopTimer();

            } else {

                startTimer();

            }

        });

 

        resetBtn.addEventListener('click', resetTimer);
